# LME Shell runtime requirements for Fedora 44, x86_64.
# This is a human-readable DNF package list, not a pip requirements file.
# Quickshell needs version 0.3.2 or newer with PipeWire, Polkit and Wayland/BackgroundEffect support.
# Availability of quickshell in Fedora 44 enabled repositories must be checked
# locally. Upstream also documents the errornointernet/quickshell COPR fallback.

[required-shell]
quickshell
python3
qt6-qtbase
qt6-qtdeclarative
qt6-qtwayland
qt6-qtsvg
qt6-qtimageformats
google-noto-sans-fonts
google-roboto-fonts
google-noto-color-emoji-fonts
adwaita-icon-theme

[required-desktop-integration]
# niri >= 25.11; full-screen popup blur additionally needs niri >= 26.04.
niri
NetworkManager
pipewire
pipewire-pulseaudio
wireplumber
brightnessctl
bluez
systemd
util-linux
polkit
wl-clipboard
# Clipboard history uses Python stdlib SQLite; cliphist is not needed.
# util-linux supplies rfkill for scoped Bluetooth software-block recovery.

[power-profile-provider-choose-one]
# Fedora's existing tuned-ppd is suitable. Do not install both providers.
tuned-ppd
# Alternative only if you deliberately use it instead of tuned-ppd:
power-profiles-daemon

[optional-tools]
# Existing preferred alternatives can be configured in the shell settings.
kitty
nautilus
firefox
pavucontrol
network-manager-applet
blueman
# Required for session locking and enabled laptop idle profiles:
swaylock
swayidle
# Needed for Night Light and local screenshot OCR respectively:
wlsunset
tesseract
tesseract-langpack-eng
tesseract-langpack-vie
libnotify
playerctl
xdg-utils
xdg-desktop-portal
xdg-desktop-portal-gnome
xdg-desktop-portal-gtk
gnome-keyring
# Optional wallpaper daemon. Check repository availability before installing.
# Upstream: https://codeberg.org/LGFae/awww.git (client and awww-daemon).
awww
# Optional external screenshot utilities. Screenshot studio uses niri capture,
# Qt Canvas annotations and Tesseract OCR; it needs neither grim nor slurp.
grim
slurp

# System monitor and keybind viewer use Python stdlib only.
# Native popup blur uses Quickshell Wayland BackgroundEffect; no capture/shader package.
